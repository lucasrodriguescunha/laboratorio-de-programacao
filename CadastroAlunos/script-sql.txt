-- 1. BANCO DE DADOS ---------------------------------------------------

CREATE DATABASE IF NOT EXISTS escola
    DEFAULT CHARACTER SET utf8mb4;

USE escola;


-- 2. TABELA ALUNO (pedida no enunciado) -------------------------------

CREATE TABLE IF NOT EXISTS aluno (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  email VARCHAR(100) NOT NULL
);


-- 3. TABELA PROFESSOR (nova) ------------------------------------------

CREATE TABLE IF NOT EXISTS professor (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  disciplina VARCHAR(100) NOT NULL
);


-- 4. TABELA CURSO (nova) ----------------------------------------------
-- carga_horaria é o total de horas do curso e precisa ser maior que zero.

CREATE TABLE IF NOT EXISTS curso (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  carga_horaria INT NOT NULL,
  CHECK (carga_horaria > 0)
);


-- 5. COMANDOS EXECUTADOS PELO SISTEMA ---------------------------------
-- Ficam no código das telas (pacote cadastroalunos.telas). Cada "?" é
-- preenchido pelo PreparedStatement com o valor digitado no campo.
--
--   Tela de Alunos       (botão Salvar)  INSERT INTO aluno (nome, email) VALUES (?, ?)
--   Tela de Alunos       (botão Listar)  SELECT id, nome, email FROM aluno ORDER BY id
--   Cadastro de Professor (botão Inserir) INSERT INTO professor (nome, disciplina) VALUES (?, ?)
--   Listagem de Professores (botão Listar) SELECT id, nome, disciplina FROM professor ORDER BY id
--   Cadastro de Curso    (botão Inserir) INSERT INTO curso (nome, carga_horaria) VALUES (?, ?)
--   Listagem de Cursos   (botão Listar)  SELECT id, nome, carga_horaria FROM curso ORDER BY id


-- 6. CONSULTAS PARA CONFERIR OS DADOS GRAVADOS PELO SISTEMA -----------

SELECT * FROM aluno;
SELECT * FROM professor;
SELECT * FROM curso;
